# Multi-Platform Merge Logic
SELECT t1.order_id, t1.date, t2.sessions, t3.ad_spend
FROM `shopify.orders` AS t1
JOIN `ga4.sessions` AS t2 ON t1.date = t2.date
JOIN `meta.ads` AS t3 ON t1.date = t3.date

LOAD INTO `bigquery.master_table`;
