   489→function Counter({ target, prefix = '', suffix = '' }: { target: number; prefix?: string; suffix?: string }) {
   490→  const [count, setCount] = useState(0);
   491→  const ref = useRef<HTMLSpanElement>(null);
   492→  const hasAnimated = useRef(false);
   493→
   494→  useEffect(() => {
   495→    const el = ref.current;
   496→    if (!el) return;
   497→    const runAnimation = () => {
   498→      if (hasAnimated.current) return;
   499→      hasAnimated.current = true;
   500→      const duration = 2000;
   501→      const start = performance.now();
   502→      const animate = (now: number) => {
   503→        const elapsed = now - start;
   504→        const progress = Math.min(elapsed / duration, 1);
   505→        const eased = 1 - Math.pow(1 - progress, 3);
   506→        setCount(Math.floor(eased * target));
   507→        if (progress < 1) requestAnimationFrame(animate);
   508→      };
   509→      requestAnimationFrame(animate);
   510→    };
   511→    const observer = new IntersectionObserver(
   512→      ([entry]) => {
   513→        if (entry.isIntersecting) {
   514→          runAnimation();
   515→        }
   516→      },
   517→      { threshold: 0.1 }
   518→    );
   519→    observer.observe(el);
   520→    // Fallback: start animation after 3s even if observer doesn't fire
   521→    const fallback = setTimeout(runAnimation, 3000);
   522→    return () => {
   523→      observer.disconnect();
   524→      clearTimeout(fallback);
   525→    };
   526→  }, [target]);
   527→
   528→  return <span ref={ref}>{prefix}{count.toLocaleString()}{suffix}</span>;
   529→}
